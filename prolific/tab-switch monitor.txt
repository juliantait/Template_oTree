    // ── AI-safety / tab-switch monitor ───────────────────────────────────────
    // Activates only after the participant has agreed to the AI-safety page,
    // which sets sessionStorage.aiSafetyAgreed = '1'. Server-side authority is
    // participant.ai_safety_disqualified, set by the focus_live_method live
    // handler bound to each monitored oTree page.
// CREDITED TO NICOLAS ORLINK
    //
    // Keep these constants in sync with intro/__init__.py.
    const AI_SAFETY = {
        MAX_VIOLATIONS: 2,        // disqualify on the 2nd recorded violation
        THRESHOLD_MS: 4000,       // continuous away-time before a violation is recorded
        OVERLAY_DELAY_MS: 400,    // grace window before the red overlay appears, to
                                  // avoid a visible flash during normal page navigation
    };

    function aiSafetyArmed() {
        try { return sessionStorage.getItem('aiSafetyAgreed') === '1'; }
        catch (e) { return false; }
    }

    // Outro pages (final survey, payment summary, thank-you, disqualified) are
    // not part of the monitored experiment. We disable the monitor entirely so
    // that participants who tab away to copy a Prolific code don't get warned.
    function inUnmonitoredSection() {
        try {
            return /\/outro\//i.test(window.location.pathname);
        } catch (e) { return false; }
    }

    function buildTabMonitorDom() {
        if (document.getElementById('tabmon-overlay')) return;

        // Red full-screen overlay (countdown visible while away)
        const overlay = document.createElement('div');
        overlay.id = 'tabmon-overlay';
        overlay.style.cssText = [
            'display:none', 'position:fixed', 'inset:0', 'z-index:9000',
            'background:rgba(198,40,40,0.92)', 'color:white',
            'flex-direction:column', 'align-items:center', 'justify-content:center',
            'text-align:center', 'padding:24px',
            'font-family:Arial, Helvetica, sans-serif',
            'backdrop-filter:blur(2px)', 'cursor:pointer',
        ].join(';');
        overlay.innerHTML = ''
          + '<div style="background:rgba(255,255,255,0.08); border:2px solid white;'
          + ' border-radius:12px; padding:28px 36px; max-width:520px;">'
          +   '<p style="font-size:24px; font-weight:bold; margin:0 0 10px 0;">Return to the study</p>'
          +   '<p style="font-size:15px; margin:0 0 14px 0;">The study page is no longer active.<br>'
          +     'Return within <strong>4&nbsp;seconds</strong> to avoid a recorded violation.</p>'
          +   '<div id="tabmon-countdown" style="font-size:52px; font-weight:bold; line-height:1;">4</div>'
          + '</div>';
        document.body.appendChild(overlay);

        // Warning modal shown after the first recorded violation
        const modal = document.createElement('div');
        modal.id = 'tabmon-modal';
        modal.style.cssText = [
            'display:none', 'position:fixed', 'inset:0', 'z-index:9999',
            'background:rgba(0,0,0,0.75)',
            'align-items:center', 'justify-content:center', 'padding:20px',
            'font-family:Arial, Helvetica, sans-serif',
        ].join(';');
        modal.innerHTML = ''
          + '<div style="background:white; color:#1a1a2e; padding:28px 32px;'
          + ' border-radius:10px; max-width:480px; text-align:center;">'
          +   '<p id="tabmon-modal-text" style="font-size:15px; line-height:1.55; margin:0 0 18px 0;"></p>'
          +   '<button id="tabmon-modal-btn" style="padding:10px 26px; background:#1a1a2e;'
          +     ' color:white; border:none; border-radius:4px; font-size:15px; cursor:pointer;">'
          +     'Understood — continue</button>'
          + '</div>';
        document.body.appendChild(modal);

        document.getElementById('tabmon-modal-btn').addEventListener('click', function () {
            modal.style.display = 'none';
            window._tabmonModalOpen = false;
        });
    }

    function startTabMonitor() {
        // Stop on outro pages even if the sessionStorage flag is still set —
        // the final survey is post-experiment and must not trigger warnings.
        if (inUnmonitoredSection()) {
            try { sessionStorage.removeItem('aiSafetyAgreed'); } catch (e) {}
            return;
        }
        if (!aiSafetyArmed() || window._tabmonStarted) return;
        window._tabmonStarted = true;
        buildTabMonitorDom();

        const overlay = document.getElementById('tabmon-overlay');
        const modal = document.getElementById('tabmon-modal');
        const countdownEl = document.getElementById('tabmon-countdown');
        const modalText = document.getElementById('tabmon-modal-text');

        // Restore prior count for in-page banner / display only. Server is
        // authoritative on disqualification.
        let count = parseInt(sessionStorage.getItem('focusLossCount') || '0', 10);
        let leaveTimer = null;
        let overlayTimer = null;
        let countdownInterval = null;
        let overlayVisible = false;
        let clickedInside = false;
        // isNavigatingAway: set when the participant submits a form or the
        // browser is unloading the page for normal navigation. Suppresses the
        // overlay/violation flow so a normal "Next" click doesn't flash red.
        let isNavigatingAway = false;
        window._tabmonModalOpen = false;

        function showOverlay() {
            if (overlayVisible || isNavigatingAway) return;
            overlayVisible = true;
            overlay.style.display = 'flex';
            // Countdown starts from the full threshold even though the overlay
            // appears OVERLAY_DELAY_MS late — the actual violation timer below
            // still fires at THRESHOLD_MS from the original blur.
            let remaining = Math.ceil(
                (AI_SAFETY.THRESHOLD_MS - AI_SAFETY.OVERLAY_DELAY_MS) / 1000
            );
            countdownEl.textContent = remaining;
            countdownInterval = setInterval(function () {
                remaining -= 1;
                countdownEl.textContent = Math.max(remaining, 0);
                if (remaining <= 0) {
                    clearInterval(countdownInterval);
                    countdownInterval = null;
                }
            }, 1000);
        }

        function hideOverlay() {
            if (!overlayVisible) return;
            overlayVisible = false;
            overlay.style.display = 'none';
            if (countdownInterval) { clearInterval(countdownInterval); countdownInterval = null; }
        }

        function showModal(text) {
            window._tabmonModalOpen = true;
            modalText.textContent = text;
            modal.style.display = 'flex';
        }

        function recordViolation() {
            hideOverlay();
            if (window._tabmonModalOpen) return;
            count += 1;
            try { sessionStorage.setItem('focusLossCount', String(count)); } catch (e) {}
            // Send to server (authoritative). live_method binds on each monitored
            // page; if liveSend isn't available (race during load) we still keep
            // the local count so the next page can resync.
            // Unique event_id is the server-side dedup key. Each real focus
            // loss generates exactly one event_id; the server increments the
            // authoritative count once per new event_id seen.
            const event_id = Date.now().toString(36) + '-' +
                Math.random().toString(36).slice(2, 10);
            if (typeof liveSend === 'function') {
                try {
                    liveSend({
                        type: 'focus_loss',
                        event_id: event_id,
                        count: count,
                        timestamp: Date.now(),
                        page: window.location.pathname,
                    });
                } catch (e) {}
            }
            if (count < AI_SAFETY.MAX_VIOLATIONS) {
                showModal(
                    'Warning ' + count + ' of ' + AI_SAFETY.MAX_VIOLATIONS +
                    ': we recorded that the study tab was inactive. One more such ' +
                    'event will end your participation and you may no longer be ' +
                    'eligible for payment or bonus compensation.'
                );
            }
            // Note: actual redirect on the disqualifying violation is driven by
            // the liveRecv handler below, so the redirect only fires once the
            // server has flagged ai_safety_disqualified.
        }

        function startLeaveTimer() {
            if (!aiSafetyArmed()) return;
            // Normal navigation (form submit, page unload) suppresses the
            // monitor so a Next click doesn't briefly flash the red overlay.
            if (isNavigatingAway) return;
            if (leaveTimer || window._tabmonModalOpen) return;
            // Delay the overlay so harmless blur events during page navigation
            // do not produce a visible red flash. The violation timer still
            // fires at the full THRESHOLD_MS from the original blur.
            overlayTimer = setTimeout(function () {
                overlayTimer = null;
                showOverlay();
            }, AI_SAFETY.OVERLAY_DELAY_MS);
            leaveTimer = setTimeout(function () {
                leaveTimer = null;
                recordViolation();
            }, AI_SAFETY.THRESHOLD_MS);
        }

        function cancelLeaveTimer() {
            if (overlayTimer) { clearTimeout(overlayTimer); overlayTimer = null; }
            if (leaveTimer)   { clearTimeout(leaveTimer);   leaveTimer   = null; }
            hideOverlay();
        }

        overlay.addEventListener('click', function () {
            cancelLeaveTimer();
        });

        // Avoid false positives caused by in-page clicks that briefly drop window focus.
        document.addEventListener('mousedown', function () { clickedInside = true; });
        document.addEventListener('mouseup',   function () { setTimeout(function () { clickedInside = false; }, 300); });

        window.addEventListener('blur', function () {
            if (clickedInside) return;
            startLeaveTimer();
        });
        window.addEventListener('focus', function () { cancelLeaveTimer(); });

        // visibilitychange fires for tab switches and OS-level window hiding;
        // blur covers cross-window focus only. Both end up calling the same
        // startLeaveTimer, which guards against double-counting via leaveTimer.
        document.addEventListener('visibilitychange', function () {
            if (document.hidden) startLeaveTimer();
            else cancelLeaveTimer();
        });

        // Server-driven redirect on disqualification. oTree's liveRecv calls
        // window.liveRecv with the live_method's per-player payload.
        window.liveRecv = function (data) {
            if (data && data.action === 'disqualified') {
                try { sessionStorage.removeItem('aiSafetyAgreed'); } catch (e) {}
                isNavigatingAway = true;
                cancelLeaveTimer();
                // Reload triggers oTree's is_displayed chain; all monitored
                // pages return False once ai_safety_disqualified is set, so
                // the participant lands on the Disqualified page in outro.
                window.location.reload();
            }
        };

        // Normal navigation should not trigger the warning. Mark intent on
        // submit/unload so a transient blur during navigation is ignored.
        function markNavigatingAway() {
            isNavigatingAway = true;
            cancelLeaveTimer();
        }
        document.addEventListener('submit', markNavigatingAway, true);
        window.addEventListener('beforeunload', markNavigatingAway);
        window.addEventListener('pagehide', markNavigatingAway);
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', startTabMonitor);
    } else {
        startTabMonitor();
    }
